sp_finalizar_compra junto com sp_cancelar_pedido - este código mostra as triggers de estoque agindo, porque o estoque baixa na compra e volta no cancelamento:
BEGIN TRAN
DECLARE @p INT
SELECT id, estoque FROM produtos WHERE id IN (2,7)                 -- antes: 5 e 5

EXEC sp_finalizar_compra @cliente_id=1, @endereco_id=1,
     @itens_json='[{"produtoId":2,"quantidade":1},{"produtoId":7,"quantidade":2}]',
     @forma_pagamento='CARTAO_CREDITO', @quantidade_parcelas=3, @valor_frete=20.00,
     @bandeira='visa', @final_cartao='1234', @pedido_id=@p OUTPUT

SELECT id, estoque FROM produtos WHERE id IN (2,7)                 -- depois: 4 e 3 (trigger baixou)
SELECT id, total, status FROM pedidos WHERE id = @p                -- status 2 (trigger confirmou o pagamento)
SELECT * FROM pagamentos WHERE pedido_id = @p                      -- 3 parcelas (fn_parcelas)

EXEC sp_cancelar_pedido @pedido_id=@p, @cliente_id=1
SELECT id, estoque FROM produtos WHERE id IN (2,7)                 -- voltou para 5 e 5 (trigger devolveu)
SELECT id, status FROM pedidos WHERE id = @p                       -- status 4 (cancelado)
ROLLBACK

-- tr_pagamentos_confirma_pedido: pagamento aprovado muda o pedido de 1 para 2
BEGIN TRAN
SELECT id, status FROM pedidos WHERE id = 81                       -- 1 (aguardando)
UPDATE pagamentos SET status = 2 WHERE pedido_id = 81
SELECT id, status FROM pedidos WHERE id = 81                       -- 2 (processando)
ROLLBACK

-- tr_entregas_conclui_pedido: entrega entregue conclui o pedido (2 para 3)
BEGIN TRAN
SELECT id, status FROM pedidos WHERE id = 123                      -- 2
UPDATE entregas SET status = 4 WHERE id = (SELECT entrega_id FROM pedidos WHERE id = 123)
SELECT id, status FROM pedidos WHERE id = 123                      -- 3
ROLLBACK
